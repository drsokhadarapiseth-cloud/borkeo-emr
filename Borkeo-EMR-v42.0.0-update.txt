================================================================================
Borkeo EMR Update Note: Version 42.0.0
Superuser works as each staff role - side menu click fix - IPD transfer page fix
================================================================================

Date: 2026-10-21
Previous Version: 41.9.0
New Version: 42.0.0
Patient data: not touched. No database change.

--------------------------------------------------------------------------------
1. WHAT WAS WRONG
--------------------------------------------------------------------------------
a) Clicking "Register" in the side menu went back to Home.
   The side menu is collapsed until the mouse moves onto it. When it expanded,
   the section headings grew by 28 px and pushed every item down, so the click
   landed on the item above: Register opened Home, Triage opened Register.
   This happened for every account, not only the Superuser.

b) The Superuser used a separate "SUPERUSER" role that had to be added by hand
   to every role check. Some were missed: the Triage queue and the OPD queue
   were refused (error 403).

c) The IPD transfer page (/ipd-transfer) did not work for anyone: its script
   and stylesheet were never served (404).

--------------------------------------------------------------------------------
2. WHAT CHANGED
--------------------------------------------------------------------------------
a) Side menu: section headings keep the same height when the menu is collapsed
   or expanded (the collapsed menu still shows a thin line). Items never move.

b) Superuser becomes the staff role of the page that is opened:
     Home                         -> Superuser (all-access overview)
     Register, Appointments       -> Registration
     Triage, Triage queue display -> Triage
     OPD queue, IPD transfer      -> Doctor (OPD)
     Paraclinic handoffs          -> Nurse (OPD)
     Medical record, Wards        -> Doctor (patient's ward, or current ward,
                                     or Medicine)
     Nursing record               -> Nurse (Midwife in Obstetric)
     Laboratory, Lab materials    -> Laboratory
     Imaging, Films               -> Imaging
     Pharmacy pages               -> Pharmacy
     Billing                      -> Cashier
     Expenses                     -> Accountant
     Cash flow                    -> Director
     Requests                     -> Admin (Director is kept)
     Administration pages         -> Admin
   If the current role already fits the page it is kept (for example Nurse on
   Daily ward medicines, Accountant on Billing). The Patient list keeps the
   current role. The role shown in the user menu is the role in use, and the
   user menu can still switch role by hand.
   Because the Superuser now holds the real role, every existing rule works
   the same as for that staff member, and one person can take a patient
   through every step and approve their own requests.
   Every automatic switch is written to the audit log (WORK_CONTEXT_CHANGED,
   automatic: true, with the page).

c) /public/ipd-transfer.js and /public/ipd-transfer.css are now served.

Checked in a test copy with synthetic patients only:
  register -> triage to OPD -> cashier payment -> OPD queue -> OPD record ->
  lab -> pharmacy; and register -> triage to Medicine ward -> wards and beds ->
  medical record (doctor) -> nursing record (nurse) -> home.
  Every side menu item was clicked in Microsoft Edge with the menu collapsed.
  npm test: 903 of 903 pass. npm run check:khmer passes.

+function emrRules(database, params) {
+  const view = String(params.get('view') || '').toLowerCase();
+  if (view === 'register') return [['REGISTRATION']];
+  if (view === 'triage') return [['TRIAGE']];
+  if (view === 'opd' || view === 'opd-paraclinic') return [['DOCTOR', 'opd']];
+  const key = params.get('patient') || params.get('patientKey') || '';
+  const row = key ? database.findPatient(key) : null;
+  const ward = row ? String(patientWorkWard(row) || '').toLowerCase() : '';
+  // Without a view the server already routes each role to its own part of
+  // the record, so any role that can open it is kept.
+  const keep = view || !row ? [] : [['REGISTRATION'], ['TRIAGE'], ['SUPERUSER'], ...MANAGEMENT_RULES];
+  if (row && ward === 'opd') return [['DOCTOR', 'opd'], ['NURSE', 'opd'], ...keep];
+  const place = ward || '!opd';
+  const nursing = ward === 'obstetric' ? [['MIDWIFE', place]] : [['NURSE', place], ['MIDWIFE', place]];
+  if (view === 'nursing') return nursing;
+  return [['DOCTOR', place], ['CHIEF_WARD', place], ...nursing, ...keep];

+function pageRules(database, pathname, params) {
+  if (pathname === '/emr') return emrRules(database, params);

+function wardFits(rule, ward) {
+  if (!rule) return !ward;
+  if (rule === '*') return Boolean(ward);
+  if (rule.startsWith('!')) return Boolean(ward) && ward !== rule.slice(1);
+  return ward === rule;
+}

+function usable(access, role, ward, database) {
+  if (!access.roles.includes(role)) return false;
+  if (!roleRequiresWard(role)) return !ward;
+  return access.wards.includes(ward) && roleWardAllowed(role, ward, database);
+}
+
+function defaultWard(access, role, rule, currentWard, database) {
+  if (!roleRequiresWard(role)) return '';
+  const fits = ward => wardFits(rule, ward) && usable(access, role, ward, database);
+  if (fits(currentWard)) return currentWard;
+  if (fits('medicine')) return 'medicine';
+  return access.wards.find(fits) || null;
+}

+// Returns the { role, wardCode } to switch to, or null when the current
+// context already fits the page (or the page has no owner role).
+export function superuserPageContext(database, session, pathname, params) {
+  const rules = pageRules(database, pathname, params);
+  if (!rules) return null;
+  const access = getUserAccess(database, session.user_id);
+  const role = sessionRole(session);
+  const ward = sessionWard(session);
+  if (rules.some(([ruleRole, rule]) => ruleRole === role && wardFits(rule, ward) && usable(access, role, ward, database))) return null;
+  for (const [ruleRole, rule] of rules) {
+    if (!access.roles.includes(ruleRole)) continue;
+    const wardCode = defaultWard(access, ruleRole, rule, ward, database);
+    if (wardCode !== null) return { role: ruleRole, wardCode };
+  }
+  return null;

+    };
+    const open = async (pathname, dest = 'document') => {

+      return response.status;
+    };

+

+    // that background request must not undo the chosen role.
